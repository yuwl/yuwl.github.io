1. MySQL 基础入门

本指南涵盖了 MySQL 中最常用的 DDL、DML、DQL 和 DCL 语句。

1. DDL (Data Definition Language) - 数据定义语言​

用于定义或修改数据库的结构（数据库、表、索引等）。

数据库操作​

CREATE DATABASE db_name;             -- 创建数据库
DROP DATABASE db_name;               -- 删除数据库
SHOW DATABASES;                      -- 查看所有数据库
USE db_name;                         -- 切换数据库

表操作​

-- 创建表
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL,
    email VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 查看表结构
show create table users;

-- 修改表结构
ALTER TABLE users ADD age INT;              -- 添加字段
ALTER TABLE users MODIFY age TINYINT;       -- 修改字段类型
ALTER TABLE users DROP COLUMN email;        -- 删除字段
RENAME TABLE old_name TO new_name;          -- 重命名表
DROP TABLE users;                           -- 删除表
TRUNCATE TABLE users;                       -- 清空表数据（重置自增 ID）

-- 索引操作 (DDL)
CREATE INDEX idx_username ON users(username);           -- 创建普通索引
CREATE UNIQUE INDEX uk_email ON users(email);           -- 创建唯一索引
ALTER TABLE users ADD INDEX idx_age(age);               -- 通过 ALTER 添加索引
DROP INDEX idx_username ON users;                       -- 删除索引
SHOW INDEX FROM users;                                  -- 查看表索引

2. DML (Data Manipulation Language) - 数据操作语言​

用于对表中的数据进行增、删、改。

-- 插入数据
INSERT INTO users (username, age) VALUES ('Root', 25);
INSERT INTO users (username, age) VALUES ('Admin', 30), ('Guest', 20); -- 批量操作

-- 更新数据
UPDATE users SET age = 26 WHERE username = 'Root';

-- 删除数据
DELETE FROM users WHERE id = 1;

3. DQL (Data Query Language) - 数据查询语言​

用于查询数据（最常用）。

-- 基本查询
SELECT * FROM users;
SELECT DISTINCT age FROM users;                  -- 去重查询

-- 条件过滤
SELECT * FROM users WHERE age > 18 AND age < 30;
SELECT * FROM users WHERE username LIKE 'A%';     -- 模糊查询 (A开头)
SELECT * FROM users WHERE age IN (20, 25, 30);

-- 排序与分页
SELECT * FROM users ORDER BY age DESC, id ASC;   -- 多重排序
SELECT * FROM users LIMIT 0, 10;                 -- 分页 (偏移量, 数量)

-- 聚合查询
SELECT COUNT(*) FROM users;
SELECT age, COUNT(*) FROM users GROUP BY age HAVING COUNT(*) > 1;

4. DCL (Data Control Language) - 数据控制语言​

用于定义数据库的访问权限和安全级别。

-- 创建用户
CREATE USER 'username'@'localhost' IDENTIFIED BY 'password';

-- 授权
GRANT ALL PRIVILEGES ON db_name.* TO 'username'@'localhost';

-- 撤销权限
REVOKE DELETE ON db_name.* FROM 'username'@'localhost';

-- 删除用户
DROP USER 'username'@'localhost';

5. 视图 (View)​

视图是基于查询结果的虚拟表，不存储实际数据。

-- 创建视图
CREATE VIEW user_emails AS
SELECT id, username, email FROM users WHERE email IS NOT NULL;

-- 使用视图
SELECT * FROM user_emails;

-- 修改视图
CREATE OR REPLACE VIEW user_emails AS
SELECT id, username, email, age FROM users;

-- 删除视图
DROP VIEW user_emails;

6. 存储过程 (Stored Procedure)​

存储过程是一组为了完成特定功能的 SQL 语句集。

-- 创建存储过程
DELIMITER //
CREATE PROCEDURE GetUserByAge(IN user_age INT)
BEGIN
    SELECT * FROM users WHERE age = user_age;
END //
DELIMITER ;

-- 调用存储过程
CALL GetUserByAge(25);

-- 删除存储过程
DROP PROCEDURE IF EXISTS GetUserByAge;

7. 运维相关命令​

用于数据库状态查看、进程管理及数据备份。

状态与进程​

SHOW PROCESSLIST;                     -- 查看当前所有连接进程
KILL process_id;                      -- 杀死某个卡住的进程
SHOW STATUS LIKE 'Threads_connected'; -- 查看当前连接数
SHOW VARIABLES LIKE 'max_connections';-- 查看最大连接数配置

数据备份与恢复 (命令行执行)​

# 备份整个数据库
mysqldump -u root -p db_name > backup.sql

# 备份特定表
mysqldump -u root -p db_name table_name > table_backup.sql

# 恢复数据
mysql -u root -p db_name < backup.sql

数据库调优分析​

EXPLAIN SELECT * FROM users WHERE id = 1; -- 分析 SQL 执行计划
SHOW PROFILE;                             -- 查看最近一条 SQL 的耗时细节 (需配置开启)
